 23    @staticmethod
 24    def pipeline_to_state(pipeline: Pipeline) -> WorkflowState:
 25        """Convert a Pipeline domain object to a LangGraph state dict.
 26
 27        Args:
 28            pipeline: Domain pipeline aggregate root.
 29
 30        Returns:
 31            Partial WorkflowState with pipeline fields populated.
 32        """
 33        gate = pipeline.last_gate_decision
 34        current_stage = pipeline.stages.get(pipeline.current_position)
 35
 36        state = WorkflowState(
 37            pipeline_id=pipeline.pipeline_id,
 38            pipeline_status=pipeline.status.value,
 39            current_position=pipeline.current_position,
 40            last_gate_decision=gate.value if gate is not None else None,
 41        )
 42
 43        if current_stage:
 44            state["current_stage_id"] = current_stage.stage_id
 45            state["stage_status"] = current_stage.status.value
 46            state["iteration_count"] = current_stage.iteration_count
 47
 48        return state

 50    @staticmethod
 51    def state_to_pipeline(state: WorkflowState) -> Pipeline:
 52        """Reconstruct a Pipeline domain object from LangGraph state.
 53
 54        Args:
 55            state: LangGraph state dictionary.
 56
 57        Returns:
 58            Pipeline domain object reflecting the stored state.
 59        """
 60        gate_str = state.get("last_gate_decision")
 61        gate = GateDecision(gate_str) if gate_str else None
 62
 63        pipeline = Pipeline(
 64            pipeline_id=state["pipeline_id"],
 65            current_position=state.get("current_position", "phase0"),
 66            status=PipelineStatus(state.get("pipeline_status", "not_started")),
 67            last_gate_decision=gate,
 68        )
 69
 70        # Restore current stage state if present
 71        current_stage_id = state.get("current_stage_id")
 72        if current_stage_id and current_stage_id in pipeline.stages:
 73            stage = pipeline.stages[current_stage_id]
 74            stage.status = StageStatus(str(state.get("stage_status") or "pending"))
 75            stage.iteration_count = state.get("iteration_count", 0)
 76
 77        return pipeline

def node_auto_gate( state: WorkflowState) -> WorkflowState: 
65def node_auto_gate(state: WorkflowState) -> WorkflowState:
66    """DAG node: Evaluate auto-gate and record decision.

def node_orchestrator( state: WorkflowState) -> WorkflowState: 
181def node_orchestrator(state: WorkflowState) -> WorkflowState:
182    """DAG node: Master orchestrator entrypoint for phases and stages.
183
184    Corresponds to FR-002, FR-017, FR-018.
185    """
186    pipeline = StateMapper.state_to_pipeline(state)
187    metadata = state.get("metadata", {})
188
189    # Delegate to Orchestrator based on current state
190    if pipeline.status == PipelineStatus.NOT_STARTED:
191        result = Orchestrator.execute_phase(0, metadata)
192    else:
193        pos = pipeline.current_position
194        stage = pos.get("stage", 0) if isinstance(pos, dict) else 0
195        result = Orchestrator.execute_stage(stage, metadata)
196
197    metadata["orchestrator_result"] = result
198    return {"metadata": metadata}

def should_continue_iterating( state: WorkflowState) -> str: 
126def should_continue_iterating(state: WorkflowState) -> str:
127    """LangGraph conditional edge: decide whether to iterate or gate.
128
129    Returns "iterate" if the current stage has not converged,
130    "gate" if it has reached its fixed point.
131
132    Args:
133        state: Current LangGraph workflow state.
134
135    Returns:
136        Edge name: "iterate" | "gate"
137    """
138    stage = StateMapper.state_to_stage(state)
139    if stage is None:
140        return "gate"
141    # Fixed-point reached if max_iterations hit or stage is passed
142    if stage.status == StageStatus.PASSED:
143        return "gate"
144    if stage.iteration_count >= MAX_ITERATIONS:
145        return "gate"
146    return "iterate"
