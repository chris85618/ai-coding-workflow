32class IterationGraphBuilder:
33    """ALG-001: Builds the Agent α/β Dual-Agent Iteration Loop subgraph.
34
35    Wraps MicroValidationGraphBuilder as a nested subgraph.
36    Includes conditional edges for convergence and HITL gate routing.
37    """
38
39    @classmethod
40    def build(cls) -> CompiledGraph:
41        """Build and compile the dual-agent iteration subgraph.
42
43        Returns:
44            Compiled LangGraph application for the α/β iteration loop.
45        """
46        graph = StateGraph(WorkflowState)
47
48        mv_app = MicroValidationGraphBuilder.build()
49
50        graph.add_node("alpha", agent_alpha_critique)
51        graph.add_node("beta", agent_beta_resolve)
52        graph.add_node("micro_val", mv_app)  # Subgraph invocation
53        graph.add_node("rca", root_cause_leftshift)
54
55        graph.set_entry_point("alpha")
56        graph.add_conditional_edges("alpha", check_fixed_point, {"beta": "beta", "exit_loop": END})
57        graph.add_edge("beta", "micro_val")
58        graph.add_edge("micro_val", "rca")
59
60        graph.add_conditional_edges("rca", hitl_gate_choice, {"alpha": "alpha", "pass": END})
61        return graph.compile()
