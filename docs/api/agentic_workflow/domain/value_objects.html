27    def prune_to_budget(self, budget: int) -> RepoMap:
28        """Return a new RepoMap pruned to fit within token budget.
29
30        Args:
31            budget: Maximum token count allowed.
32
33        Returns:
34            New RepoMap with symbols trimmed to fit budget.
35        """
36        if budget <= 0:
37            return RepoMap(symbols=(), token_count=0, file_ranks={})
38
39        selected: list[SymbolDef] = []
40        token_count = 0
41        for sym in self.symbols:
42            cost = max(1, len(sym.signature) // _CHARS_PER_TOKEN)
43            if token_count + cost > budget:
44                break
45            selected.append(sym)
46            token_count += cost
47
48        return RepoMap(
49            symbols=tuple(selected),
50            token_count=token_count,
51            file_ranks=self.file_ranks,
52        )
53
54    def get_context_string(self) -> str:
55        """Render this map as a string for LLM context injection.
56
57        Returns:
58            Formatted string of symbol signatures.
59        """
60        if not self.symbols:
61            return ""
62        lines = [f"# Repository Map ({self.token_count} tokens)"]
63        current_file = ""
64        for sym in self.symbols:
65            if sym.file_path != current_file:
66                lines.append(f"\n## {sym.file_path}")
67                current_file = sym.file_path
68            lines.append(f"  {sym.kind}: {sym.signature}")
69        return "\n".join(lines)
